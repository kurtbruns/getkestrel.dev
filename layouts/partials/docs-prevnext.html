{{- /* Previous and Next stay within the page's section, so the main path ends at its last step instead of running on into the guides after it. */ -}} {{- $pages := (site.GetPage "/docs").Pages.ByWeight -}} {{- with .Params.docsection }}{{ $pages = where $pages "Params.docsection" . }}{{ end -}} {{- $cur := .RelPermalink -}} {{- $prev := false -}}{{- $next := false -}} {{- range $i, $p := $pages -}} {{- if eq $p.RelPermalink $cur -}} {{- if gt $i 0 -}}{{- $prev = index $pages (sub $i 1) -}}{{- end -}} {{- if lt (add $i 1) (len $pages) -}}{{- $next = index $pages (add $i 1) -}}{{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{ with $prev }}← Previous{{ .Title }}{{ else }}{{ end }} {{ with $next }}Next →{{ .Title }}{{ end }}
{{- end -}}
